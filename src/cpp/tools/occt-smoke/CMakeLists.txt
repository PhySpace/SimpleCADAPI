cmake_minimum_required(VERSION 3.20)
project(scad_kernel_check LANGUAGES CXX)
find_package(OpenCASCADE CONFIG REQUIRED)
add_executable(kernel_check main.cxx)
target_compile_features(kernel_check PRIVATE cxx_std_17)
target_include_directories(kernel_check PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_libraries(kernel_check PRIVATE TKFillet TKOffset)
enable_testing()
add_test(NAME patched-kernel COMMAND kernel_check)
