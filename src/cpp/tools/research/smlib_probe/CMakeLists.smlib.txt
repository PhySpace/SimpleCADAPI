cmake_minimum_required(VERSION 3.20)
project(smlib_try CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SRC /tmp/usdbrep/repo/source)
file(GLOB SMLIB_SRC ${SRC}/SMLib/src/*.cpp)
file(GLOB SMAPI_SRC ${SRC}/SM_API/src/*.cpp)
list(FILTER SMLIB_SRC EXCLUDE REGEX "StdAfx.cpp$")
list(FILTER SMAPI_SRC EXCLUDE REGEX "StdAfx.cpp$")
find_package(TBB REQUIRED)
add_library(smlib STATIC ${SMLIB_SRC})
target_include_directories(smlib PUBLIC ${SRC}/SMLib/inc)
target_compile_options(smlib PRIVATE -w)
target_link_libraries(smlib PUBLIC TBB::tbb TBB::tbbmalloc)
add_library(smapi STATIC ${SMAPI_SRC})
target_include_directories(smapi PUBLIC ${SRC}/SM_API/inc)
target_compile_options(smapi PRIVATE -w -Wno-write-strings)
target_link_libraries(smapi PUBLIC smlib)
